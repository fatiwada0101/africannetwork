### 2. Example Schemas

```javascript
// lib/schemas/purchase.js
import { z } from 'zod';

export const purchaseSchema = z.object({
  plan_id: z.string().min(1).max(100),
  payment_method: z.enum(['wallet', 'card', 'quick_buy']),
  amount: z.number().positive().max(1_000_000),
});

// lib/schemas/plans.js
export const createPlanSchema = z.object({
  name: z.string().min(1).max(100).transform(sanitizeHtml),
  price: z.number().positive().max(1_000_000),
  duration: z.string().regex(/^\d+[hdwm]$/), // 1h, 1d, 7d, 30d
  speed: z.string().max(20).optional(),
  devices: z.number().int().min(1).max(10).default(1),
  upload_speed: z.string().regex(/^\d+[MK]$/).default('12M'),
  download_speed: z.string().regex(/^\d+[MK]$/).default('12M'),
});
```

### 3. Validation Helper

```javascript
// lib/validate.js
export function validateBody(schema) {
  return async (request) => {
    try {
      const body = await request.json();
      const result = schema.safeParse(body);
      if (!result.success) {
        return { error: result.error.flatten(), data: null };
      }
      return { error: null, data: result.data };
    } catch {
      return { error: { message: 'Invalid JSON body' }, data: null };
    }
  };
}
```

### 4. HTML Sanitization

```javascript
// lib/sanitize.js
export function sanitizeHtml(input) {
  return input
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#x27;');
}
```
